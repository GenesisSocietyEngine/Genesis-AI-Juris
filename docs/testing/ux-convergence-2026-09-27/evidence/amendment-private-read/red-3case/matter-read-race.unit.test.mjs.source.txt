function load(file){
  if(overridden.has(file))return overridden.get(file);
  if(cache.has(file))return cache.get(file).exports;
  if(extname(file)==='.json')return JSON.parse(readFileSync(file,'utf8'));
  const module={exports:{}};cache.set(file,module);
  const source=readFileSync(file,'utf8');
  const compiled=ts.transpileModule(source,{fileName:file,compilerOptions:{target:ts.ScriptTarget.ES2022,module:ts.ModuleKind.CommonJS,esModuleInterop:true}}).outputText;
  function localRequire(name){
    if(name==='cloudflare:workers')return {env:{}};
    if(name==='next/headers')return {headers:async()=>new Headers()};
    if(!name.startsWith('.'))return requireExternal(name);
    const base=resolve(dirname(file),name);
    const found=[base,base+'.ts',base+'.tsx',base+'.json',resolve(base,'index.ts')].find(p=>existsSync(p)&&extname(p));
    if(!found)throw new Error('Unresolved local module: '+name);
    return load(found);
  }
  new Function('require','module','exports','__filename','__dirname',compiled)(localRequire,module,module.exports,file,dirname(file));
  return module.exports;
}

test('regression: detail response must use current membership after awaited private read',async()=>{
  reset();let started,release;
  const reached=new Promise(resolve=>{started=resolve;});

  barrier={started,resume};

  assert.ok(authorityReads>=2,'Production organization authority resolved before barrier');
  assert.equal(accessReads,1,'Production joined Matter access read completed before barrier');
  membership={...membership,status:'suspended',revision:2};
  release();

  observations.push({name:'suspended_during_read',status:response.status,membershipStatus:membership.status,membershipRevision:membership.revision,privateAssertionPresent:JSON.stringify(body).includes(statement),privateExcerptPresent:JSON.stringify(body).includes(excerpt),oldWritePermission:body.dossier?.permissions?.can_write??null,authorityReads,accessReads});

});
