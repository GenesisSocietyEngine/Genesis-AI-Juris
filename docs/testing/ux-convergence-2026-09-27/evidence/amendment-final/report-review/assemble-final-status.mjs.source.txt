const under = (child, base) => {
  const rel = path.relative(base, child);
  return rel !== '' && rel !== '..' && !rel.startsWith('..' + path.sep) && !path.isAbsolute(rel);
};
function localFile(input) {
  assert.equal(typeof input, 'string', 'Missing input path');
  const absolute = path.resolve(ROOT, input);
  assert.ok(under(absolute, ROOT), 'Input must be a file in the canonical checkout');
  const real = fs.realpathSync(absolute);
  assert.ok(under(real, fs.realpathSync(ROOT)), 'Input resolves outside canonical checkout');
  assert.ok(fs.statSync(real).isFile(), 'Input is not a file');
  return { absolute: real, bytes: fs.readFileSync(real) };
}
function jsonInput(input) {
  const f = localFile(input);
  return { ...f, hash: sha(f.bytes), value: JSON.parse(f.bytes.toString('utf8').replace(/^\uFEFF/, '')) };
}

    const key = values[i].replace(/^--/, '');

    assert.ok(values[i + 1] && !values[i + 1].startsWith('--'), 'Missing flag value');
    result[key] = values[i + 1];
  }
  return result;
}

function paragraph(text, start, replacement) {
  const blocks = text.split('\n\n'), ids = blocks.flatMap((b,i)=>b.startsWith(start)?[i]:[]);
  assert.equal(ids.length, 1, `Missing/duplicate template paragraph: ${start}`);
  blocks[ids[0]] = replacement;
  return blocks.join('\n\n');
}
function uniqueLine(text, start, replacement) {
  assert.ok(!replacement.includes('\n'), 'Line replacement contains newlines');
  const lines = text.split('\n'), ids = lines.flatMap((line,i)=>line.startsWith(start)?[i]:[]);
  assert.equal(ids.length, 1, `Missing/duplicate template line: ${start}`);
  lines[ids[0]] = replacement;
  return lines.join('\n');
}
function exact(text, old, next) {
  assert.ok(text.includes(old), `Template phrase absent: ${old}`);
  return text.replaceAll(old,next);
}
