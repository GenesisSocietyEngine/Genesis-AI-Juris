import {readFileSync,writeFileSync,mkdirSync} from 'node:fs';
import {resolve} from 'node:path';
import {createHash} from 'node:crypto';
import {spawnSync} from 'node:child_process';
const root=resolve(import.meta.dirname,'../..');
const harness='.artifacts/ux-reconciliation-auth/local-read-smoke.mjs';
const hash=bytes=>createHash('sha256').update(bytes).digest('hex');
const expectedSha256='4374127f667065141e1b77c902ab47f4910bff11ee2361ad740391e73a2a4ec5';
if(hash(readFileSync(resolve(root,harness)))!==expectedSha256)throw new Error('Reviewed harness changed; do not run.');
if(process.version!=='v22.23.2')throw new Error('Pinned Node version required.');
const args=[harness,'--run-approved','--origin=http://localhost:5281','--build-receipt=.artifacts/amendment-release-gate/candidate-initial-source.json','--expected-commit=e6c6adc6fe01129b2e3d442072c532e08591fae9','--expected-input-digest=5597958a0a870757e75cfbdfba930007d475f782035db4ee8503a97aa1723afc'];
const startedAt=new Date().toISOString();
const output=resolve(root,'.artifacts/ux-reconciliation-auth/local-read-smoke-execution-'+startedAt.replaceAll(':','').replaceAll('.',''));
mkdirSync(output,{recursive:true});
const result=spawnSync(process.execPath,args,{cwd:root,encoding:null,maxBuffer:4*1024*1024,windowsHide:true});
const completedAt=new Date().toISOString();
writeFileSync(resolve(output,'stdout.log'),result.stdout??Buffer.alloc(0));
writeFileSync(resolve(output,'stderr.log'),result.stderr??Buffer.alloc(0));
const receipt={schema:'genesis.juris.local-read-smoke-execution.v1',startedAt,completedAt,executable:process.execPath,node:process.version,args,workingDirectory:root,harnessSha256:expectedSha256,exitCode:result.status,signal:result.signal,spawnFailed:Boolean(result.error),stdout:{bytes:result.stdout?.length??0,sha256:hash(result.stdout??Buffer.alloc(0))},stderr:{bytes:result.stderr?.length??0,sha256:hash(result.stderr??Buffer.alloc(0))},scope:'Approved sequential ordinary local API smoke; script-owned login/logout sessions only. Does not prove an in-flight revocation race.'};
writeFileSync(resolve(output,'execution.json'),JSON.stringify(receipt,null,2)+'\n');
process.stdout.write(result.stdout??Buffer.alloc(0));
console.log(JSON.stringify({executionStatus:result.status===0?'PASS':'FAIL',exitCode:result.status,startedAt,completedAt,executionReceipt:output+'/execution.json'}));
if(result.status!==0)process.exitCode=1;
